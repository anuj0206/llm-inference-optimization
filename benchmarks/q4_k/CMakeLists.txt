cmake_minimum_required(VERSION 3.25)
project(q4_microbench LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
if(NOT STOCK_SOURCE OR NOT STOCK_GENERATED OR NOT STOCK_RUNTIME_BUILD)
  message(FATAL_ERROR "Supply STOCK_SOURCE, STOCK_GENERATED and STOCK_RUNTIME_BUILD")
endif()
option(Q4_TRAFFIC "Build the in-process CUPTI counter benchmark" OFF)
set(benchmark_target q4-microbench)
if(Q4_TRAFFIC)
  set(benchmark_target q4-traffic)
endif()

add_executable(${benchmark_target} benchmark.cu)
target_compile_features(${benchmark_target} PRIVATE cxx_std_17 cuda_std_17)
target_compile_definitions(${benchmark_target} PRIVATE GGML_SHARED)
target_include_directories(${benchmark_target} PRIVATE
  "${STOCK_GENERATED}" "${STOCK_SOURCE}/ggml/src/ggml-cuda"
  "${STOCK_SOURCE}/ggml/src" "${STOCK_SOURCE}/ggml/include"
  "${STOCK_SOURCE}/vendor")
target_compile_options(${benchmark_target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math;-Xcompiler=/openmp;-Xcompiler=/Zc:preprocessor;-Xcompiler=/Gy>)
target_link_options(${benchmark_target} PRIVATE /OPT:REF)
target_link_libraries(${benchmark_target} PRIVATE CUDA::cudart "${STOCK_RUNTIME_BUILD}/ggml/src/ggml-base.lib")
set_target_properties(${benchmark_target} PROPERTIES CUDA_ARCHITECTURES 89-real)

if(Q4_TRAFFIC)
  set(cupti_samples "${CUDAToolkit_BIN_DIR}/../samples/common")
  if(NOT TARGET CUDA::cupti OR NOT EXISTS "${cupti_samples}/cupti_range_profiler_util.h")
    message(FATAL_ERROR "Install the cuda_cupti component before building the counter benchmark")
  endif()
  target_compile_definitions(${benchmark_target} PRIVATE Q4_TRAFFIC)
  target_include_directories(${benchmark_target} PRIVATE "${cupti_samples}")
  target_link_libraries(${benchmark_target} PRIVATE CUDA::cupti CUDA::cuda_driver)
endif()
